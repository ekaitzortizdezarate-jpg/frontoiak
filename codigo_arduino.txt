#include <WiFi.h>
#include <WiFiClientSecure.h>
#include <HTTPClient.h>
#include <Preferences.h>
#include "time.h"

// ================= CONFIGURACIÓN BÁSICA DEL USUARIO =================
const char* WIFI_SSID         = "TU_WIFI_SSID";
const char* WIFI_PASS         = "TU_WIFI_PASSWORD";
const char* SUPABASE_URL      = "https://tu-proyecto.supabase.co";
const char* SUPABASE_ANON_KEY = "TU_SUPABASE_ANON_KEY";
const char* HARDWARE_TOKEN    = "00000000-0000-0000-0000-000000000000";

// ================= PINES DE HARDWARE =================
#define PIN_MIC     33                  // OUT del MAX9814
#define PIN_BATERIA 34                  // Divisor 100k / 100k

const char* NTP_SERVER = "pool.ntp.org";
const char* TZ_INFO    = "CET-1CEST,M3.5.0,M10.5.0/3";

// Objeto de almacenamiento Flash no volátil (NVS)
Preferences preferences;

// Variables de calibración activas en memoria
String g_modo              = "normal";
int    g_umbralSonido      = 1200;
int    g_golpesMinimos     = 3;
int    g_muestreoSeg       = 20;
int    g_dormirSeg         = 270;
int    g_reboteMs          = 180;
String g_horaInicioActiva  = "07:00";
String g_horaFinActiva     = "23:30";

// Cargar configuración guardada en memoria Flash NVS
void cargarConfiguracionFlash() {
  preferences.begin("fronton_iot", false);
  g_modo             = preferences.getString("modo", "normal");
  g_umbralSonido     = preferences.getInt("umbral", 1200);
  g_golpesMinimos    = preferences.getInt("golpes", 3);
  g_muestreoSeg      = preferences.getInt("muestreo", 20);
  g_dormirSeg        = preferences.getInt("dormir", 270);
  g_reboteMs         = preferences.getInt("rebote", 180);
  g_horaInicioActiva = preferences.getString("h_inicio", "07:00");
  g_horaFinActiva    = preferences.getString("h_fin", "23:30");
  preferences.end();
}

// Guardar nueva configuración en Flash NVS
void guardarConfiguracionFlash() {
  preferences.begin("fronton_iot", false);
  preferences.putString("modo", g_modo);
  preferences.putInt("umbral", g_umbralSonido);
  preferences.putInt("golpes", g_golpesMinimos);
  preferences.putInt("muestreo", g_muestreoSeg);
  preferences.putInt("dormir", g_dormirSeg);
  preferences.putInt("rebote", g_reboteMs);
  preferences.putString("h_inicio", g_horaInicioActiva);
  preferences.putString("h_fin", g_horaFinActiva);
  preferences.end();
  Serial.println(">> Configuracion guardada con exito en Flash NVS.");
}

// Lee el voltaje real de la batería Li-ion 18650 (3.3V a 4.2V)
float leerVoltajeBateria() {
  uint32_t sumaRaw = 0;
  for (int i = 0; i < 20; i++) {
    sumaRaw += analogRead(PIN_BATERIA);
    delay(2);
  }
  float rawPromedio = sumaRaw / 20.0;

  // Con atenuación ADC de 11dB (rango ~0V a 3.3V) y resolución de 12 bits (0 a 4095):
  // Voltaje en el pin = raw * (3.3V / 4095)
  // Multiplicamos por 2 por el divisor resistivo (100k / 100k):
  float voltajePin = (rawPromedio / 4095.0) * 3.3;
  float voltajeBateria = voltajePin * 2.0;

  Serial.printf("DEBUG BATERIA -> Raw ADC: %.1f | Voltaje Pin: %.2fV | Voltaje Calculado: %.2fV\n",
                rawPromedio, voltajePin, voltajeBateria);

  return voltajeBateria;
}

int calcularPorcentajeBateria(float v) {
  if (v >= 4.20) return 100;
  if (v <= 3.30) return 0;
  return (int)((v - 3.30) / (4.20 - 3.30) * 100.0);
}

// Escucha activa de impactos acústicos y registro de amplitud máxima registrada
int contarPelotazos(int ventanaSeg, int umbral, int reboteMs, int &amplitudMaxRegistrada) {
  int impactos = 0;
  amplitudMaxRegistrada = 0;
  unsigned long inicio = millis();
  unsigned long ultimoImpacto = 0;

  Serial.printf("Escuchando impactos (%ds, umbral: %d, rebote: %dms)...\n", ventanaSeg, umbral, reboteMs);

  while (millis() - inicio < (ventanaSeg * 1000UL)) {
    unsigned long ventana = millis();
    int senalMax = 0;
    int senalMin = 4095;

    while (millis() - ventana < 20) {
      int lectura = analogRead(PIN_MIC);
      if (lectura > senalMax) senalMax = lectura;
      if (lectura < senalMin) senalMin = lectura;
    }

    int amplitudPico = senalMax - senalMin;
    if (amplitudPico > amplitudMaxRegistrada) {
      amplitudMaxRegistrada = amplitudPico;
    }

    if (amplitudPico > umbral && (millis() - ultimoImpacto > (unsigned long)reboteMs)) {
      impactos++;
      ultimoImpacto = millis();
      Serial.printf("Pelotazo detectado! (#%d) - Pico: %d | AmplitudMax: %d\n", impactos, amplitudPico, amplitudMaxRegistrada);
    }
  }
  return impactos;
}

// Extraer valor de clave JSON sencilla (para compatibilidad total sin dependencias externas)
String extraerJsonString(const String& json, const String& clave) {
  String patron = "\"" + clave + "\":\"";
  int idx = json.indexOf(patron);
  if (idx == -1) return "";
  idx += patron.length();
  int fin = json.indexOf("\"", idx);
  if (fin == -1) return "";
  return json.substring(idx, fin);
}

int extraerJsonInt(const String& json, const String& clave, int valorDefecto) {
  String patron = "\"" + clave + "\":";
  int idx = json.indexOf(patron);
  if (idx == -1) return valorDefecto;
  idx += patron.length();
  while (idx < (int)json.length() && (json[idx] == ' ' || json[idx] == '\"')) idx++;
  int fin = idx;
  while (fin < (int)json.length() && (isDigit(json[fin]) || json[fin] == '-')) fin++;
  if (fin > idx) {
    return json.substring(idx, fin).toInt();
  }
  return valorDefecto;
}

// Analizar la respuesta remota del RPC de Supabase y actualizar variables si hay cambios
void procesarRespuestaConfig(const String& respuestaJson) {
  if (respuestaJson.indexOf("\"config\":") == -1) return;

  bool cambios = false;

  String nuevoModo = extraerJsonString(respuestaJson, "modo");
  if (nuevoModo.length() > 0 && nuevoModo != g_modo) {
    g_modo = nuevoModo;
    cambios = true;
  }

  int nuevoUmbral = extraerJsonInt(respuestaJson, "umbral_sonido", g_umbralSonido);
  if (nuevoUmbral != g_umbralSonido) {
    g_umbralSonido = nuevoUmbral;
    cambios = true;
  }

  int nuevosGolpes = extraerJsonInt(respuestaJson, "golpes_minimos", g_golpesMinimos);
  if (nuevosGolpes != g_golpesMinimos) {
    g_golpesMinimos = nuevosGolpes;
    cambios = true;
  }

  int nuevoMuestreo = extraerJsonInt(respuestaJson, "muestreo_seg", g_muestreoSeg);
  if (nuevoMuestreo != g_muestreoSeg) {
    g_muestreoSeg = nuevoMuestreo;
    cambios = true;
  }

  int nuevoDormir = extraerJsonInt(respuestaJson, "dormir_seg", g_dormirSeg);
  if (nuevoDormir != g_dormirSeg) {
    g_dormirSeg = nuevoDormir;
    cambios = true;
  }

  int nuevoRebote = extraerJsonInt(respuestaJson, "rebote_ms", g_reboteMs);
  if (nuevoRebote != g_reboteMs) {
    g_reboteMs = nuevoRebote;
    cambios = true;
  }

  String nuevaHoraInicio = extraerJsonString(respuestaJson, "hora_inicio_activa");
  if (nuevaHoraInicio.length() >= 5 && nuevaHoraInicio != g_horaInicioActiva) {
    g_horaInicioActiva = nuevaHoraInicio.substring(0, 5);
    cambios = true;
  }

  String nuevaHoraFin = extraerJsonString(respuestaJson, "hora_fin_activa");
  if (nuevaHoraFin.length() >= 5 && nuevaHoraFin != g_horaFinActiva) {
    g_horaFinActiva = nuevaHoraFin.substring(0, 5);
    cambios = true;
  }

  if (cambios) {
    Serial.printf(">> Nueva configuracion remota recibida: modo=%s, umbral=%d, dormir=%ds\n",
                  g_modo.c_str(), g_umbralSonido, g_dormirSeg);
    guardarConfiguracionFlash();
  }
}

// Verifica si la hora actual está dentro de la franja operativa activa
bool estaEnHorarioActivo(int horaActual, int minutoActual) {
  int minutosActuales = horaActual * 60 + minutoActual;

  int hInicio = g_horaInicioActiva.substring(0, 2).toInt();
  int mInicio = g_horaInicioActiva.substring(3, 5).toInt();
  int minutosInicio = hInicio * 60 + mInicio;

  int hFin = g_horaFinActiva.substring(0, 2).toInt();
  int mFin = g_horaFinActiva.substring(3, 5).toInt();
  int minutosFin = hFin * 60 + mFin;

  if (minutosFin >= minutosInicio) {
    return (minutosActuales >= minutosInicio && minutosActuales <= minutosFin);
  } else {
    // Cruza medianoche
    return (minutosActuales >= minutosInicio || minutosActuales <= minutosFin);
  }
}

// Enviar datos por HTTPS llamando al RPC `registrar_pulso_iot`
void enviarASupabase(bool enUso, float batVoltaje, int batPorcentaje, int golpes, int amplitudMax, int &horaOut, int &minOut) {
  if (WiFi.status() != WL_CONNECTED) return;

  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) {
    Serial.println("Error obteniendo hora NTP");
    return;
  }

  horaOut = timeinfo.tm_hour;
  minOut  = timeinfo.tm_min;

  char fecha[11], horaInicio[9], horaFin[9];
  strftime(fecha, sizeof(fecha), "%Y-%m-%d", &timeinfo);

  // Redondear a tramos de 5 minutos exactos (00, 05, 10, ... 55)
  int minutoBase = (timeinfo.tm_min / 5) * 5;
  int minutoFin  = minutoBase + 5;
  int horaFinVal = timeinfo.tm_hour;

  if (minutoFin >= 60) {
    minutoFin = 0;
    horaFinVal = (horaFinVal + 1) % 24;
  }

  snprintf(horaInicio, sizeof(horaInicio), "%02d:%02d:00", timeinfo.tm_hour, minutoBase);
  snprintf(horaFin, sizeof(horaFin), "%02d:%02d:00", horaFinVal, minutoFin);

  WiFiClientSecure client;
  client.setInsecure();

  HTTPClient https;
  String endpoint = String(SUPABASE_URL) + "/rest/v1/rpc/registrar_pulso_iot";

  if (https.begin(client, endpoint)) {
    https.addHeader("Content-Type", "application/json");
    https.addHeader("apikey", SUPABASE_ANON_KEY);
    https.addHeader("Authorization", String("Bearer ") + SUPABASE_ANON_KEY);

    String payload = "{";
    payload += "\"p_token\":\"" + String(HARDWARE_TOKEN) + "\",";
    payload += "\"p_en_uso\":" + String(enUso ? "true" : "false") + ",";
    payload += "\"p_fecha\":\"" + String(fecha) + "\",";
    payload += "\"p_hora_inicio\":\"" + String(horaInicio) + "\",";
    payload += "\"p_hora_fin\":\"" + String(horaFin) + "\",";
    payload += "\"p_bateria_pct\":" + String(batPorcentaje) + ",";
    payload += "\"p_bateria_v\":" + String(batVoltaje, 2) + ",";
    payload += "\"p_golpes\":" + String(golpes) + ",";
    payload += "\"p_amplitud_max\":" + String(amplitudMax);
    payload += "}";

    int httpCode = https.POST(payload);
    String respuesta = https.getString();
    Serial.printf("Supabase HTTP %d: %s\n", httpCode, respuesta.c_str());

    if (httpCode >= 200 && httpCode < 300) {
      procesarRespuestaConfig(respuesta);
    }
    https.end();
  }
}

void setup() {
  Serial.begin(115200);
  analogReadResolution(12);
  analogSetPinAttenuation(PIN_BATERIA, ADC_11db); // Escala completa 0V - 3.3V en el pin 34

  // 1. Cargar parámetros guardados en memoria no volátil
  cargarConfiguracionFlash();

  // 2. Determinar ventana de muestreo según el modo activo
  int ventanaMuestreo = (g_modo == "config") ? 5 : g_muestreoSeg;

  // 3. Medir batería y escuchar cancha antes de iniciar radio WiFi
  float batV = leerVoltajeBateria();
  int batPct = calcularPorcentajeBateria(batV);
  int ampMax = 0;
  int totalGolpes = contarPelotazos(ventanaMuestreo, g_umbralSonido, g_reboteMs, ampMax);
  bool frontonOcupado = (totalGolpes >= g_golpesMinimos);

  Serial.printf("Modo: %s | Bateria: %.2fV (%d%%) | Golpes: %d | AmpMax: %d -> %s\n",
                g_modo.c_str(), batV, batPct, totalGolpes, ampMax, frontonOcupado ? "EN USO" : "LIBRE");

  // 4. Conectar al WiFi y sincronizar con Supabase
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  int intentos = 0;
  while (WiFi.status() != WL_CONNECTED && intentos < 20) {
    delay(500);
    intentos++;
  }

  int horaActual = 12, minActual = 0;
  if (WiFi.status() == WL_CONNECTED) {
    configTzTime(TZ_INFO, NTP_SERVER);
    enviarASupabase(frontonOcupado, batV, batPct, totalGolpes, ampMax, horaActual, minActual);
  }

  // 5. Determinar tiempo de sueño según modo y horario activo
  uint64_t segundosDormir = g_dormirSeg;

  if (g_modo == "config") {
    // En modo configuración y calibración: ciclo rápido de 5 segundos
    segundosDormir = 5;
    Serial.println(">> Modo Config: Durmiendo 5 segundos para calibracion rapida...");
  } else {
    // En modo normal: verificar si estamos fuera de horario activo
    if (!estaEnHorarioActivo(horaActual, minActual)) {
      segundosDormir = 600; // 10 minutos de reposo profundo nocturno
      Serial.println(">> Fuera de horario activo: Durmiendo 600 segundos (modo ahorro)...");
    } else {
      Serial.printf(">> Modo Normal: Durmiendo %llu segundos...\n", segundosDormir);
    }
  }

  // 6. Entrar en Deep Sleep
  WiFi.disconnect(true);
  WiFi.mode(WIFI_OFF);
  esp_sleep_enable_timer_wakeup(segundosDormir * 1000000ULL);
  esp_deep_sleep_start();
}
